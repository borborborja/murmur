cmake_minimum_required(VERSION 3.22)
project(murmur_whisper LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(WHISPER_CURL OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ALL_VARIANTS OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ARM_ARCH armv8-a CACHE STRING "" FORCE)
add_subdirectory(../.inference-cache/whisper.cpp whisper)
add_library(murmur_whisper SHARED murmur_whisper.cpp)
target_include_directories(murmur_whisper PRIVATE ../.inference-cache/whisper.cpp/examples)
target_link_libraries(murmur_whisper PRIVATE whisper log)
target_link_options(murmur_whisper PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,--exclude-libs,ALL")
set_target_properties(murmur_whisper PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
